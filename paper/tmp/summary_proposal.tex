% ==================== 3. PROBLEM STATEMENT & CONTEXT ====================
\section{Phát biểu Bài toán \& Ngữ cảnh}

\begin{frame}{Phát biểu Bài toán \& Ngữ cảnh}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{Vấn đề}
\small
Các bộ giả lập SDC phải thực thi hàng ngàn kịch bản đường đi trong mỗi phiên bản, tiêu tốn \textbf{hàng ngàn giờ tính toán}. Do đó, chúng ta phải \emph{ưu tiên} sao cho các \textbf{kịch bản lỗi} được mô phỏng đầu tiên.
\end{block}

\begin{block}{Đầu vào \& Đầu ra}
\small
\textbf{Đầu vào}: Cấu trúc hình học của đường $\mathcal{R} = \{p_1, \ldots, p_N\} \subset \mathbb{R}^2$ \\
\textbf{Đầu ra}: Hoán vị xếp hạng $\pi^*$ của $n$ kịch bản kiểm thử dựa trên xác suất dự đoán xảy ra lỗi.
\end{block}

\column{0.48\textwidth}
\centering
\includegraphics[width=\linewidth, keepaspectratio]{../slides/latex_proposal/example.png}\\[0.3em]
\scriptsize\textit{BeamNG.tech: PASS (màu xanh) vs FAIL (màu đỏ)}

\vspace{0.5em}
\begin{alertblock}{Ràng buộc \& Thách thức}
\scriptsize
\begin{itemize}
    \item \textbf{Độ đo}: Tối ưu hoá \textbf{APFD} (Tỷ lệ trung bình các lỗi được phát hiện) -- \emph{không thể vi phân}.
    \item \textbf{Tính đối xứng}: Các điểm số dự đoán phải bất biến với nhóm SE(2).
\end{itemize}
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Vì sao các phương pháp hiện tại thất bại -- Hai điểm mù}
\vspace{1em}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{1. Sự nhạy cảm khi lấy mẫu}
\small
Cùng một con đường, 64 so với 197 điểm $\Rightarrow$ điểm số khác nhau.\\[4pt]
$\Delta\text{APFD} \approx 0.04\text{--}0.07$\\[4pt]
Phụ thuộc vào mức độ rời rạc hoá.
\end{alertblock}

\column{0.48\textwidth}
\begin{alertblock}{2. Phụ thuộc vào hệ quy chiếu}
\small
Xoay đường đi $30^\circ$ $\Rightarrow$ APFD giảm 4--7 điểm.\\[4pt]
$f(R \cdot \mathcal{R}) \neq f(\mathcal{R})$\\[4pt]
Vi phạm trực giác vật lý.
\end{alertblock}
\end{columns}
\end{frame}

% ==================== 4. METHODOLOGY ====================
\section{Phương pháp luận / Đề xuất mở rộng}

\begin{frame}{Phương pháp luận: Hướng tiếp cận mới}
\begin{block}{Giải quyết bài toán từ góc độ Hình học}
\small
\textbf{Nguyên lý vật lý}: Sự kiện "xe chệch khỏi làn đường" phụ thuộc vào \textbf{hình dạng nội tại} của con đường, độc lập với toạ độ tuyệt đối $(x,y)$ hay góc hướng $\theta$. Do đó, chúng tôi \textbf{loại bỏ hoàn toàn} các thuộc tính phụ thuộc vào hệ quy chiếu.

\textbf{Giải pháp}: Xây dựng \textbf{tập đặc trưng 7 kênh} bất biến với SE(2):
\vspace{-0.3em}
\begin{columns}[T]
\column{0.45\textwidth}
\begin{itemize}
    \item Chiều dài phân đoạn ($\Delta s$)
    \item Độ cong có dấu ($\kappa$)
    \item Đạo hàm độ cong ($\Delta \kappa$)
    \item Đạo hàm bậc hai ($\Delta^2 \kappa$)
\end{itemize}
\column{0.5\textwidth}
\begin{itemize}
    \item Chiều dài cung tương đối ($s_{norm}$)
    \item Thay đổi góc cục bộ ($|\Delta \theta|$)
    \item Độ lệch chuẩn cục bộ (local std)
\end{itemize}
\end{columns}
\end{block}
\end{frame}

\begin{block}{Định lý đúng hoàn toàn về mặt số học (Bit-Identically)}
\scriptsize
Với mọi $(R, t) \in \mathrm{SE}(2)$: \quad $f_{\theta}(R \cdot \mathcal{R} + t) = f_{\theta}(\mathcal{R})$

Điểm số, thứ hạng và APFD là \alert{hoàn toàn bằng nhau} -- không phải xấp xỉ. Chỉ dùng kỹ thuật tăng cường dữ liệu (data augmentation) sẽ không thể đảm bảo sự tuân thủ chặt chẽ nguyên lý vật lý này.
\end{block}
\end{frame}

\section{Kiến trúc \& Thuật toán}

\begin{frame}[fragile]{Kiến trúc \& Thuật toán cốt lõi}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Pipeline: SE2RoadNet}
\scriptsize
\begin{enumerate}
    \item \textbf{Đầu vào}: Các điểm đường đi $(x,y)$.
    \item \textbf{Trích xuất đặc trưng}: Tính toán 7 kênh bất biến SE(2) (độ cong, chiều dài cung).
    \item \textbf{Backbone}: Transformer 6 lớp với positional bias dựa trên chiều dài cung tương đối.
    \item \textbf{Hàm mất mát}: Focal BCE.
\end{enumerate}
\end{block}

\begin{exampleblock}{Tư tưởng của Thuật toán}
\scriptsize
Bằng cách loại bỏ toạ độ tuyệt đối $(x,y)$ và hướng toàn cục, mô hình buộc phải dựa \textbf{hoàn toàn vào hình dạng nội tại của con đường}.
\end{exampleblock}

\column{0.50\textwidth}
\begin{block}{Đoạn mã: Đặc trưng bất biến}
\tiny
\begin{semiverbatim}
def extract_invariant_7ch(pts_raw):
    # pts: (N, 2). KHÔNG có x, y hay hướng thô!
    pts = np.array(pts_raw).reshape(-1,2)
    d = np.diff(pts, axis=0)
    seg = np.linalg.norm(d, axis=1) # 1. Chiều dài
    
    k = signed_curvature(pts)       # 3. Độ cong
    dk = np.diff(k)                 # 4. d(kappa)
    ddk = np.diff(dk)               # 5. d2(kappa)
    
    s_cum = np.cumsum(seg)          
    s_norm = s_cum / s_cum[-1]      # 6. Chiều dài cung TĐ
    
    # ... tính toán 7 đặc trưng bất biến ...
    return np.column_stack([
        seg, abs_dang, k, dk, ddk, s_norm, lstd
    ])
\end{semiverbatim}
\end{block}
\end{columns}
